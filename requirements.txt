# Системная инструкция для ИИ (ваши правила питания и логика)
SYSTEM_INSTRUCTION = """
Ты персональный ИИ-диетолог и кулинарный помощник. Твой подозпечный соблюдает следующие жесткие правила:
1. Строгая кето-диета (низкое количество углеводов, полезные жиры, умеренный белок).

@dp.message(CommandStart())
async def command_start_handler(message: types.Message):
    await message.answer(
        "Привет! Я твой персональный кето-помощник и диетолог. "
        "Я слежу за кашрутом, кето-правилами и твоим весом.\n\n"
        "Отправь мне текстовое описание еды или фотографию блюда, и я проведу полный анализ!"
    )
 
@dp.message(F.text)
async def handle_text(message: types.Message):
    # Отправляем текстовый запрос в Gemini с нашими правилами
    response = gemini_client.models.generate_content(
        model='gemini-2.5-flash',
        contents=message.text,
        config={
            'system_instruction': SYSTEM_INSTRUCTION
        }
    )
    await message.answer(response.text)
 
async def main():
    logging.basicConfig(level=logging.INFO, stream=sys.stdout)
    print("Бот запущен и ждет сообщения...")
    await dp.start_polling(bot)
 
if __name__ == "__main__":
    asyncio.run(main())
